cmake_minimum_required(VERSION 3.16)
project(magiccad_dwg_adapter C)
set(CMAKE_C_STANDARD 99)
if(NOT LIBREDWG_SOURCE OR NOT LIBREDWG_BUILD)
  message(FATAL_ERROR "Set LIBREDWG_SOURCE and LIBREDWG_BUILD")
endif()
find_library(REDWG_LIBRARY NAMES redwg libredwg PATHS "${LIBREDWG_BUILD}" "${LIBREDWG_BUILD}/Release" NO_DEFAULT_PATH REQUIRED)
add_executable(magiccad-dwg-write dwg-write.c)
target_include_directories(magiccad-dwg-write PRIVATE "${LIBREDWG_SOURCE}/include")
target_link_libraries(magiccad-dwg-write PRIVATE "${REDWG_LIBRARY}")
if(UNIX)
  target_link_libraries(magiccad-dwg-write PRIVATE m)
endif()
